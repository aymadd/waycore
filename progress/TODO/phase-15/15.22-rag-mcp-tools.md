All tools should return responses with:
1. **Content** - The actual information
2. **Safety level** - Warning indicator
3. **Source attribution** - Where the info came from
4. **Disclaimer** - Standard safety warning

EDIBILITY: All parts edible (leaves, roots, flowers)

Source: Plants For A Future Database

from __future__ import annotations

from mcp.server import Server
from mcp.types import Tool, TextContent

from device.services.ai_service.rag.embeddings import EmbeddingGenerator
from device.services.ai_service.rag.models import Category
from device.services.ai_service.rag.store import HybridKnowledgeStore

from .formatters import format_search_results, format_plant_result, format_first_aid

app = Server("waycore-outdoor-knowledge")

# Initialize on startup
store = HybridKnowledgeStore()
store.load_index()
embedder = EmbeddingGenerator()


@app.list_tools()

@app.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    """Handle tool calls."""

    if name == "search_survival_knowledge":
        return await handle_search(arguments)
    elif name == "identify_plant":
        return await handle_plant_id(arguments)
    elif name == "get_first_aid":
        return await handle_first_aid(arguments)
    elif name == "lookup_knot":
        return await handle_knot_lookup(arguments)
    elif name == "get_weather_signs":
        return await handle_weather(arguments)
    else:
        return [TextContent(type="text", text=f"Unknown tool: {name}")]


async def handle_search(args: dict) -> list[TextContent]:
    """Handle general survival knowledge search."""
    query = args.get("query", "")
    category = args.get("category")
    max_results = min(args.get("max_results", 5), 10)

    # Generate query embedding
    query_embedding = embedder.embed_query(query)

    # Search
    results = store.search(
        query=query,
        query_embedding=query_embedding,
        category=category,
        limit=max_results,
    )

    if not results:
        return [TextContent(
            type="text",
            text="No relevant information found in the knowledge base."
        )]

    formatted = format_search_results(results)
    return [TextContent(type="text", text=formatted)]


async def handle_plant_id(args: dict) -> list[TextContent]:
    """Handle plant identification lookup."""
    plant_name = args.get("plant_name", "")

    query_embedding = embedder.embed_query(plant_name)

    results = store.search(
        query=plant_name,
        query_embedding=query_embedding,
        category="plants",
        limit=3,
    )

    if not results:
        return [TextContent(
            type="text",
            text=(
                f"No information found for '{plant_name}'.\n\n"
                "⚠️ IMPORTANT: Do not consume any plant without positive identification "
                "from multiple reliable sources AND expert verification.\n\n"
                "Many poisonous plants closely resemble edible ones. "
                "When in doubt, DO NOT consume."
            )
        )]

    formatted = format_plant_result(results[0])
    return [TextContent(type="text", text=formatted)]


async def handle_first_aid(args: dict) -> list[TextContent]:
    """Handle first aid guidance lookup."""
    condition = args.get("condition", "")

    query_embedding = embedder.embed_query(f"first aid {condition}")

    results = store.search(
        query=condition,
        query_embedding=query_embedding,
        category="first_aid",
        limit=3,
    )

    if not results:
        return [TextContent(
            type="text",
            text=(
                f"No specific guidance found for '{condition}'.\n\n"
                "⚠️ IMPORTANT: For any serious injury or medical emergency:\n"
                "1. Call emergency services if possible\n"
                "2. Activate SOS beacon if in wilderness\n"
                "3. Keep the person calm and stable\n"
                "4. Do not move them if spinal injury suspected\n\n"
                "This is not medical advice. Seek professional care."
            )
        )]

    formatted = format_first_aid(results)
    return [TextContent(type="text", text=formatted)]


async def handle_knot_lookup(args: dict) -> list[TextContent]:
    """Handle knot lookup."""
    knot_name = args.get("knot_name", "")

    query_embedding = embedder.embed_query(f"knot {knot_name}")

    results = store.search(
        query=knot_name,
        query_embedding=query_embedding,
        category="knots",
        limit=2,
    )

    if not results:
        return [TextContent(
            type="text",
            text=f"No instructions found for '{knot_name}' knot."
        )]

    result = results[0]
    return [TextContent(
        type="text",
        text=(
            f"🪢 {result['title']}\n\n"
            f"{result['content']}\n\n"
            f"Source: {result['source_file']} p.{result['source_page']}"
        )
    )]


async def handle_weather(args: dict) -> list[TextContent]:
    """Handle weather signs lookup."""
    query = args.get("query", "")

    query_embedding = embedder.embed_query(f"weather {query}")

    results = store.search(
        query=query,
        query_embedding=query_embedding,
        category="weather",
        limit=3,
    )

    if not results:
        return [TextContent(
            type="text",
            text=f"No weather information found for '{query}'."
        )]

    formatted = ["🌦️ Weather Signs\n"]
    for r in results:
        formatted.append(f"\n{r['title']}\n{r['content'][:400]}\n")

    return [TextContent(type="text", text="".join(formatted))]

### Response Formatters

```python
from __future__ import annotations

from typing import Any


def format_search_results(results: list[dict[str, Any]]) -> str:
    """Format general search results."""
    lines = ["📚 Relevant Survival Knowledge\n"]

    for i, r in enumerate(results, 1):
        lines.append(f"\n--- {i}. {r['title']} ({r['category']}) ---")
        lines.append(r["content"][:500])

        if r["safety_level"] not in ["safe", None]:
            lines.append(f"\n⚠️ SAFETY ({r['safety_level'].upper()}): {r['safety_notes']}")

        lines.append(f"\nSource: {r['source_file']} p.{r['source_page']}")

    return "\n".join(lines)


def format_plant_result(plant: dict[str, Any]) -> str:
    """Format plant identification result with strong safety warnings."""

    safety_icons = {
        "safe": "✅",
        "caution": "⚠️",
        "warning": "⚠️",
        "danger": "🚫",
        "lethal": "☠️",
    }

    icon = safety_icons.get(plant["safety_level"], "⚠️")

    lines = [
        f"🌿 {plant['title']}",
        f"Category: {plant['subcategory']}",
        "",
        plant["content"],
        "",
        "=" * 50,
        f"{icon} SAFETY LEVEL: {plant['safety_level'].upper()}",
        plant["safety_notes"] or "Always verify identification.",
        "",
        "⚠️ CRITICAL WARNING:",
        "• Many plants have DEADLY look-alikes",
        "• NEVER consume without expert verification",
        "• When in doubt, DO NOT EAT",
        "• This is for educational purposes ONLY",
        "",
        f"Source: {plant.get('source_url', 'Knowledge base')}",
    ]

    return "\n".join(lines)


def format_first_aid(results: list[dict[str, Any]]) -> str:
    """Format first aid guidance with medical disclaimer."""

    lines = ["🏥 Wilderness First Aid Guidance\n"]

    for r in results:
        lines.append(f"\n{r['title']}")
        lines.append("-" * 40)
        lines.append(r["content"][:600])
        lines.append(f"\nSource: {r['source_file']} p.{r['source_page']}")

    lines.extend([
        "",
        "=" * 50,
        "⚠️ IMPORTANT MEDICAL DISCLAIMER:",
        "• This is EDUCATIONAL information only",
        "• NOT a substitute for professional medical care",
        "• Call emergency services when possible",
        "• Activate SOS beacon in wilderness emergencies",
        "• Do not delay seeking professional help",
    ])

    return "\n".join(lines)
```
