```python
class SafetyLevel(str, Enum):
    """Safety classification for knowledge entries."""
    SAFE = "safe"           # General info, no special warnings
    CAUTION = "caution"     # Requires care (e.g., knife handling)
    WARNING = "warning"     # Significant risk (e.g., fire starting)
    DANGER = "danger"       # High risk (e.g., plant identification)
    LETHAL = "lethal"       # Life-threatening if wrong (e.g., mushrooms)


class Category(str, Enum):
    """Knowledge categories matching Waycore use cases."""
    SURVIVAL = "survival"
    NAVIGATION = "navigation"
    FIRST_AID = "first_aid"
    PLANTS = "plants"
    WEATHER = "weather"
    KNOTS = "knots"
    SHELTER = "shelter"
    WATER = "water"
    FIRE = "fire"
    SIGNALING = "signaling"
    WILDLIFE = "wildlife"
    EQUIPMENT = "equipment"


@dataclass
class KnowledgeEntry:
    """A single searchable knowledge entry."""
    id: str
    title: str
    content: str
    category: Category
    subcategory: str = ""

    # Safety metadata
    safety_level: SafetyLevel = SafetyLevel.SAFE
    safety_notes: str = ""

    # Source tracking
    source_file: str = ""
    source_page: int = 0
    source_url: str = ""
    license: str = "public_domain"

    # Search optimization
    tags: list[str] = field(default_factory=list)
    keywords: list[str] = field(default_factory=list)

    # Embeddings (populated during indexing)
    embedding: list[float] = field(default_factory=list)
```

### PDF Parser

```python
from __future__ import annotations

import hashlib
import re
from dataclasses import dataclass
from pathlib import Path

from ..models import Category, KnowledgeEntry, SafetyLevel


class PDFParser:
    """Parse PDF documents into knowledge entries.

    Uses PyMuPDF (fitz) for text extraction - lightweight and works on ARM64.
    """

    def __init__(self, chunk_size: int = 500, overlap: int = 50):
        self.chunk_size = chunk_size
        self.overlap = overlap

    def parse_file(
        self,
        pdf_path: Path,
        category: Category,
        default_safety: SafetyLevel = SafetyLevel.SAFE,
    ) -> list[KnowledgeEntry]:
        """Parse a PDF file into knowledge entries."""
        try:
            import fitz  # PyMuPDF
        except ImportError:
            raise ImportError("PyMuPDF required: pip install pymupdf")

        entries = []
        doc = fitz.open(pdf_path)

        current_chapter = ""
        current_section = ""

        for page_num, page in enumerate(doc, start=1):
            text = page.get_text()

            # Detect chapter/section headers
            chapter_match = re.search(r"CHAPTER\s+(\d+)[:\s-]+(.+)", text, re.I)
            if chapter_match:
                current_chapter = chapter_match.group(2).strip()

            section_match = re.search(r"^([A-Z][A-Z\s]+)$", text, re.M)
            if section_match:
                current_section = section_match.group(1).strip().title()

            # Chunk the page content
            chunks = self._chunk_text(text)

            for i, chunk in enumerate(chunks):
                if len(chunk.strip()) < 50:  # Skip tiny chunks
                    continue

                entry_id = self._generate_id(pdf_path.stem, page_num, i)

                entries.append(KnowledgeEntry(
                    id=entry_id,
                    title=f"{current_chapter} - {current_section}".strip(" -"),
                    content=chunk,
                    category=category,
                    subcategory=current_section,
                    safety_level=self._detect_safety(chunk, default_safety),
                    source_file=pdf_path.name,
                    source_page=page_num,
                    tags=self._extract_tags(chunk),
                    keywords=self._extract_keywords(chunk),
                ))

        doc.close()
        return entries

    def _chunk_text(self, text: str) -> list[str]:
        """Split text into overlapping chunks."""
        words = text.split()
        chunks = []

        for i in range(0, len(words), self.chunk_size - self.overlap):
            chunk = " ".join(words[i:i + self.chunk_size])
            chunks.append(chunk)

        return chunks

    def _generate_id(self, source: str, page: int, chunk: int) -> str:
        """Generate unique ID for an entry."""
        raw = f"{source}:{page}:{chunk}"
        return hashlib.md5(raw.encode()).hexdigest()[:12]

    def _detect_safety(self, text: str, default: SafetyLevel) -> SafetyLevel:
        """Detect safety level from content."""
        text_lower = text.lower()

        # Lethal indicators
        if any(w in text_lower for w in ["poison", "toxic", "deadly", "fatal"]):
            return SafetyLevel.LETHAL

        # Danger indicators
        if any(w in text_lower for w in ["dangerous", "hazard", "venomous"]):
            return SafetyLevel.DANGER

        # Warning indicators
        if any(w in text_lower for w in ["caution", "careful", "warning", "risk"]):
            return SafetyLevel.WARNING

        return default

    def _extract_tags(self, text: str) -> list[str]:
        """Extract relevant tags from content."""
        survival_terms = {
            "fire", "water", "shelter", "food", "signal", "navigate",
            "first aid", "snake", "plant", "edible", "poison", "trap",
            "knot", "rope", "weather", "storm", "cold", "heat",
        }

        text_lower = text.lower()
        return [term for term in survival_terms if term in text_lower]

    def _extract_keywords(self, text: str) -> list[str]:
        """Extract searchable keywords."""
        stopwords = {"the", "a", "an", "is", "are", "was", "were", "be",
                     "to", "of", "and", "or", "in", "on", "at", "for"}

        words = re.findall(r'\b[a-zA-Z]{4,}\b', text.lower())
        return list(set(w for w in words if w not in stopwords))[:20]
```

### Plant Data Parser

```python
from __future__ import annotations

import csv
import json
from pathlib import Path

from ..models import Category, KnowledgeEntry, SafetyLevel


class PlantDataParser:
    """Parse plant databases into knowledge entries."""

    def parse_pfaf_json(self, json_path: Path) -> list[KnowledgeEntry]:
        """Parse Plants For A Future database export."""
        entries = []

        with open(json_path) as f:
            plants = json.load(f)

        for plant in plants:
            safety = self._determine_safety(plant)
            content = self._build_plant_content(plant)

            entries.append(KnowledgeEntry(
                id=f"plant_{plant.get('latin_name', '').replace(' ', '_').lower()}",
                title=plant.get("common_name", plant.get("latin_name", "Unknown")),
                content=content,
                category=Category.PLANTS,
                subcategory=plant.get("family", ""),
                safety_level=safety,
                safety_notes=self._build_safety_notes(plant),
                source_url="https://pfaf.org/",
                license="cc_by_nc_sa",
                tags=self._build_plant_tags(plant),
            ))

        return entries

    def parse_usda_csv(self, csv_path: Path) -> list[KnowledgeEntry]:
        """Parse USDA Plants Database CSV export."""
        entries = []

        with open(csv_path, newline='', encoding='utf-8') as f:
            reader = csv.DictReader(f)

            for row in reader:
                entries.append(KnowledgeEntry(
                    id=f"usda_{row.get('Symbol', '').lower()}",
                    title=row.get("Common Name", row.get("Scientific Name", "")),
                    content=self._build_usda_content(row),
                    category=Category.PLANTS,
                    subcategory=row.get("Family", ""),
                    safety_level=SafetyLevel.DANGER,  # Default cautious
                    safety_notes="Verify identification before any use.",
                    source_url="https://plants.usda.gov/",
                    license="public_domain",
                ))

        return entries

    def _determine_safety(self, plant: dict) -> SafetyLevel:
        """Determine safety level from plant data."""
        if plant.get("known_hazards"):
            return SafetyLevel.LETHAL

        edibility = plant.get("edibility_rating", 0)
        if edibility >= 4:
            return SafetyLevel.CAUTION

        return SafetyLevel.DANGER  # Default to cautious

    def _build_plant_content(self, plant: dict) -> str:
        """Build searchable content string."""
        parts = [
            f"Scientific name: {plant.get('latin_name', 'Unknown')}",
            f"Common name: {plant.get('common_name', 'Unknown')}",
            f"Family: {plant.get('family', 'Unknown')}",
        ]

        if plant.get("edible_parts"):
            parts.append(f"Edible parts: {plant['edible_parts']}")
        if plant.get("medicinal_uses"):
            parts.append(f"Medicinal uses: {plant['medicinal_uses']}")
        if plant.get("habitat"):
            parts.append(f"Habitat: {plant['habitat']}")

        return "\n".join(parts)

    def _build_safety_notes(self, plant: dict) -> str:
        """Build safety warning string."""
        notes = []

        if plant.get("known_hazards"):
            notes.append(f"HAZARDS: {plant['known_hazards']}")

        notes.append("Always verify identification with multiple sources.")
        notes.append("When in doubt, do NOT consume.")

        return " ".join(notes)

    def _build_plant_tags(self, plant: dict) -> list[str]:
        """Build tags for plant entry."""
        tags = []

        if plant.get("edibility_rating", 0) > 0:
            tags.append("edible")
        if plant.get("medicinal_rating", 0) > 0:
            tags.append("medicinal")
        if plant.get("known_hazards"):
            tags.append("hazardous")

        return tags
```
