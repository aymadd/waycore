## MCP Protocol Overview

The Model Context Protocol provides:

- **Tools**: Functions the AI can call (get_temperature, search_docs, etc.)
- **Resources**: Data sources the AI can read (files, databases)
- **Prompts**: Reusable prompt templates
- **Sampling**: Request completions from the AI

### MCP Message Flow

```
User Query: "What's the temperature outside?"
     │
     ▼
┌─────────────────────────────────────────────────────────────┐
│ Agent Loop                                                   │
│                                                              │
│  1. LLM analyzes query → decides to use get_temperature tool │
│  2. MCP Client sends tools/call to Sensor MCP Server        │
│  3. Server executes tool, returns result                     │
│  4. Agent incorporates result into response                  │
│  5. Return: "The current temperature is 23°C"               │
└─────────────────────────────────────────────────────────────┘
```

### Agent Controller

```python
from __future__ import annotations

import logging
from typing import Any

from device.services.ai_service.mcp.client import MCPClient
from device.services.ai_service.mcp.registry import ToolRegistry

logger = logging.getLogger(__name__)


class AgentController:
    """Controls the agentic AI loop."""

    MAX_ITERATIONS = 5  # Prevent infinite loops

    def __init__(
        self,
        llm_runner: Any,
        tool_registry: ToolRegistry,
    ) -> None:
        self.llm = llm_runner
        self.tools = tool_registry

    async def run(self, query: str, context: dict[str, Any] | None = None) -> str:
        """Run the agent loop to answer a query."""
        messages = self._build_initial_messages(query, context)

        for iteration in range(self.MAX_ITERATIONS):
            # Get LLM response
            response = await self.llm.generate(messages)

            # Check for tool calls
            tool_calls = self._extract_tool_calls(response)

            if not tool_calls:
                # No tools needed, return final response
                return response.content

            # Execute tool calls
            tool_results = await self._execute_tools(tool_calls)

            # Add results to context for next iteration
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "tool", "content": self._format_results(tool_results)})

        return "I was unable to complete this request. Please try a simpler question."

    async def _execute_tools(self, tool_calls: list[dict]) -> list[dict]:
        """Execute a list of tool calls."""
        results = []
        for call in tool_calls:
            tool_name = call["name"]
            tool_args = call.get("arguments", {})

            try:
                result = await self.tools.call(tool_name, tool_args)
                results.append({"tool": tool_name, "result": result, "error": None})
            except Exception as e:
                logger.error(f"Tool {tool_name} failed: {e}")
                results.append({"tool": tool_name, "result": None, "error": str(e)})

        return results
```

### Tool Registry

```python
from __future__ import annotations

from typing import Any, Callable

from device.services.ai_service.mcp.client import MCPClient, MCPTool


class ToolRegistry:
    """Central registry for all available tools."""

    def __init__(self) -> None:
        self._local_tools: dict[str, tuple[MCPTool, Callable]] = {}
        self._mcp_clients: dict[str, MCPClient] = {}

    def register_local(
        self,
        name: str,
        description: str,
        schema: dict[str, Any],
        handler: Callable,
    ) -> None:
        """Register a local tool (not via MCP)."""
        tool = MCPTool(name=name, description=description, input_schema=schema)
        self._local_tools[name] = (tool, handler)

    async def register_mcp_server(self, name: str, client: MCPClient) -> None:
        """Register an MCP server's tools."""
        await client.connect()
        self._mcp_clients[name] = client

    def list_tools(self) -> list[MCPTool]:
        """List all available tools."""
        tools = [t for t, _ in self._local_tools.values()]
        for client in self._mcp_clients.values():
            tools.extend(client._tools.values())
        return tools

    async def call(self, name: str, arguments: dict[str, Any]) -> Any:
        """Call a tool by name."""
        # Check local tools first
        if name in self._local_tools:
            _, handler = self._local_tools[name]
            return await handler(**arguments)

        # Check MCP servers
        for client in self._mcp_clients.values():
            if name in client._tools:
                result = await client.call_tool(name, arguments)
                return result.content

        raise ValueError(f"Unknown tool: {name}")
```
