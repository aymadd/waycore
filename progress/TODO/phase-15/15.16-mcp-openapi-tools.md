# Task: MCP OpenAPI Tools Integration

**Phase**: Phase 15 - AI App **Task ID**: 15.16 **Status**: TODO **Started**:
Not started **Completed**: Not completed **Priority**: Medium

## Description

Enable the AI agent to automatically discover and use all Waycore service APIs
by parsing the auto-generated OpenAPI specifications. This creates a dynamic
tool system where new API endpoints automatically become available to the AI.

## Current State (Problems)

1. **Manual tool creation**: Each tool must be hand-coded
2. **Maintenance burden**: API changes require tool updates
3. **Limited coverage**: Only explicitly coded tools are available
4. **No third-party support**: External apps can't expose tools

## Target State (Goals)

1. **Auto-discovery**: AI tools generated from OpenAPI specs
2. **Always current**: Tools update when APIs change
3. **Full coverage**: All API endpoints available as tools
4. **Extensible**: Third-party apps can add tools via OpenAPI

## Implementation

### OpenAPI Parser

```python
from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path
from typing import Any


@dataclass
class APIEndpoint:
    """Parsed API endpoint."""

    path: str
    method: str
    operation_id: str
    summary: str
    description: str
    parameters: list[dict[str, Any]]
    request_body: dict[str, Any] | None
    service_url: str


class OpenAPIParser:
    """Parse OpenAPI specs into tool definitions."""

    def __init__(self, spec_dir: Path) -> None:
        self.spec_dir = spec_dir

    def parse_all(self) -> list[APIEndpoint]:
        """Parse all OpenAPI specs in the directory."""
        endpoints = []

        for spec_file in self.spec_dir.glob("*.json"):
            if spec_file.name == "combined.json":
                continue  # Skip combined spec
            if spec_file.name == "ai-service.json":
                continue  # Avoid recursion

            service_endpoints = self.parse_spec(spec_file)
            endpoints.extend(service_endpoints)

        return endpoints

    def parse_spec(self, spec_path: Path) -> list[APIEndpoint]:
        """Parse a single OpenAPI spec."""
        with open(spec_path) as f:
            spec = json.load(f)

        endpoints = []
        base_url = self._get_base_url(spec, spec_path.stem)

        for path, path_item in spec.get("paths", {}).items():
            for method in ["get", "post", "put", "delete", "patch"]:
                if method not in path_item:
                    continue

                operation = path_item[method]
                endpoint = APIEndpoint(
                    path=path,
                    method=method.upper(),
                    operation_id=operation.get("operationId", f"{method}_{path}"),
                    summary=operation.get("summary", ""),
                    description=operation.get("description", ""),
                    parameters=operation.get("parameters", []),
                    request_body=operation.get("requestBody"),
                    service_url=base_url,
                )
                endpoints.append(endpoint)

        return endpoints

    def _get_base_url(self, spec: dict, service_name: str) -> str:
        """Get base URL for a service."""
        # Map service names to URLs
        service_urls = {
            "core-daemon": "http://core-daemon:8000",
            "data-logger": "http://data-logger:8002",
            "comms-bridge": "http://comms-bridge:8003",
            "sensor-hub": "http://sensor-hub:8001",
            "module-manager": "http://module-manager:8001",
        }
        return service_urls.get(service_name, f"http://{service_name}:8000")
```

### Tool Generator

```python
from __future__ import annotations

from mcp.types import Tool

from .parser import APIEndpoint


class ToolGenerator:
    """Generate MCP tools from API endpoints."""

    # Endpoints that require confirmation before execution
    DANGEROUS_OPERATIONS = {
        "factory_reset",
        "delete_all",
        "send_mesh_message",
        "clear_data",
    }

    # Endpoints to exclude from tool generation
    EXCLUDED_PATTERNS = [
        "/health",
        "/metrics",
        "/internal/",
    ]

    def generate_tools(self, endpoints: list[APIEndpoint]) -> list[Tool]:
        """Generate MCP tools from API endpoints."""
        tools = []

        for endpoint in endpoints:
            if self._should_exclude(endpoint):
                continue

            tool = self._create_tool(endpoint)
            tools.append(tool)

        return tools

    def _should_exclude(self, endpoint: APIEndpoint) -> bool:
        """Check if endpoint should be excluded."""
        for pattern in self.EXCLUDED_PATTERNS:
            if pattern in endpoint.path:
                return True
        return False

    def _create_tool(self, endpoint: APIEndpoint) -> Tool:
        """Create an MCP tool from an API endpoint."""
        # Generate tool name from operation ID
        tool_name = self._sanitize_name(endpoint.operation_id)

        # Build description
        description = endpoint.summary or endpoint.description
        if not description:
            description = f"{endpoint.method} {endpoint.path}"

        # Build input schema from parameters and request body
        input_schema = self._build_input_schema(endpoint)

        return Tool(
            name=tool_name,
            description=description,
            inputSchema=input_schema,
        )

    def _sanitize_name(self, name: str) -> str:
        """Sanitize operation ID to valid tool name."""
        # Replace non-alphanumeric with underscore
        clean = "".join(c if c.isalnum() else "_" for c in name)
        return clean.lower().strip("_")

    def _build_input_schema(self, endpoint: APIEndpoint) -> dict:
        """Build JSON schema for tool input."""
        properties = {}
        required = []

        # Add path/query parameters
        for param in endpoint.parameters:
            param_name = param["name"]
            param_schema = param.get("schema", {"type": "string"})

            properties[param_name] = {
                "type": param_schema.get("type", "string"),
                "description": param.get("description", ""),
            }

            if param.get("required", False):
                required.append(param_name)

        # Add request body properties
        if endpoint.request_body:
            content = endpoint.request_body.get("content", {})
            json_content = content.get("application/json", {})
            body_schema = json_content.get("schema", {})

            if body_schema.get("type") == "object":
                for prop_name, prop_schema in body_schema.get("properties", {}).items():
                    properties[prop_name] = prop_schema

                required.extend(body_schema.get("required", []))

        return {
            "type": "object",
            "properties": properties,
            "required": list(set(required)),
        }
```

### API Executor

```python
from __future__ import annotations

import httpx
from typing import Any

from .parser import APIEndpoint


class APIExecutor:
    """Execute API calls for tools."""

    def __init__(self, timeout: float = 10.0) -> None:
        self.timeout = timeout

    async def execute(
        self,
        endpoint: APIEndpoint,
        arguments: dict[str, Any],
    ) -> dict[str, Any]:
        """Execute an API call."""
        url = self._build_url(endpoint, arguments)
        body = self._build_body(endpoint, arguments)

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            try:
                if endpoint.method == "GET":
                    resp = await client.get(url, params=self._get_query_params(endpoint, arguments))
                elif endpoint.method == "POST":
                    resp = await client.post(url, json=body)
                elif endpoint.method == "PUT":
                    resp = await client.put(url, json=body)
                elif endpoint.method == "DELETE":
                    resp = await client.delete(url)
                else:
                    return {"error": f"Unsupported method: {endpoint.method}"}

                if resp.status_code >= 400:
                    return {"error": f"API error: {resp.status_code}", "details": resp.text}

                return resp.json() if resp.content else {"success": True}

            except Exception as e:
                return {"error": str(e)}

    def _build_url(self, endpoint: APIEndpoint, arguments: dict) -> str:
        """Build URL with path parameters."""
        url = endpoint.service_url + endpoint.path

        # Replace path parameters
        for param in endpoint.parameters:
            if param.get("in") == "path":
                param_name = param["name"]
                if param_name in arguments:
                    url = url.replace(f"{{{param_name}}}", str(arguments[param_name]))

        return url

    def _get_query_params(self, endpoint: APIEndpoint, arguments: dict) -> dict:
        """Extract query parameters from arguments."""
        query_params = {}
        for param in endpoint.parameters:
            if param.get("in") == "query":
                param_name = param["name"]
                if param_name in arguments:
                    query_params[param_name] = arguments[param_name]
        return query_params

    def _build_body(self, endpoint: APIEndpoint, arguments: dict) -> dict | None:
        """Build request body from arguments."""
        if not endpoint.request_body:
            return None

        # Remove path/query params, remaining are body params
        body = {}
        path_query_params = {p["name"] for p in endpoint.parameters}

        for key, value in arguments.items():
            if key not in path_query_params:
                body[key] = value

        return body if body else None
```

### MCP Server

```python
from __future__ import annotations

from pathlib import Path

from mcp.server import Server
from mcp.types import Tool, TextContent

from .parser import OpenAPIParser, APIEndpoint
from .generator import ToolGenerator
from .executor import APIExecutor

SPEC_DIR = Path("/app/docs/api")

app = Server("waycore-openapi")
parser = OpenAPIParser(SPEC_DIR)
generator = ToolGenerator()
executor = APIExecutor()

# Cache parsed endpoints
_endpoints: dict[str, APIEndpoint] = {}
_tools: list[Tool] = []


def refresh_tools() -> None:
    """Refresh tools from OpenAPI specs."""
    global _endpoints, _tools

    endpoints = parser.parse_all()
    _endpoints = {e.operation_id: e for e in endpoints}
    _tools = generator.generate_tools(endpoints)


# Initial load
refresh_tools()


@app.list_tools()
async def list_tools() -> list[Tool]:
    return _tools


@app.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    endpoint = _endpoints.get(name)
    if not endpoint:
        return [TextContent(type="text", text=f"Unknown tool: {name}")]

    result = await executor.execute(endpoint, arguments)

    if "error" in result:
        return [TextContent(type="text", text=f"Error: {result['error']}")]

    # Format result for AI
    import json
    formatted = json.dumps(result, indent=2)
    return [TextContent(type="text", text=formatted)]
```
