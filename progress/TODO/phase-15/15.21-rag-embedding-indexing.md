### Embedding Generator

```python
from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from .models import KnowledgeEntry


class EmbeddingGenerator:
    """Generate embeddings using sentence-transformers.

    Uses all-MiniLM-L6-v2:
    - 22MB model size (fits on RPi5)
    - 384-dimensional embeddings
    - Fast inference on ARM64
    """

    MODEL_NAME = "all-MiniLM-L6-v2"
    EMBEDDING_DIM = 384

    def __init__(self):
        self._model = None

    def _load_model(self):
        """Lazy load the model."""
        if self._model is None:
            from sentence_transformers import SentenceTransformer
            self._model = SentenceTransformer(self.MODEL_NAME)
        return self._model

    def embed_entries(
        self,
        entries: list[KnowledgeEntry],
        batch_size: int = 32,
        show_progress: bool = True,
    ) -> list[KnowledgeEntry]:
        """Add embeddings to knowledge entries."""
        model = self._load_model()

        # Combine title and content for embedding
        texts = [f"{e.title} {e.content}" for e in entries]

        embeddings = model.encode(
            texts,
            convert_to_numpy=True,
            show_progress_bar=show_progress,
            batch_size=batch_size,
        )

        for entry, embedding in zip(entries, embeddings):
            entry.embedding = embedding.tolist()

        return entries

    def embed_query(self, query: str) -> list[float]:
        """Generate embedding for a search query."""
        model = self._load_model()
        return model.encode(query, convert_to_numpy=True).tolist()
```

### Hybrid Knowledge Store

```python
from __future__ import annotations

import json
import sqlite3
from pathlib import Path
from typing import Any

from .models import KnowledgeEntry


class HybridKnowledgeStore:
    """Hybrid storage combining SQLite FTS5 and Hnswlib vector search.

    SQLite FTS5: Keyword/phrase matching, metadata filtering
    Hnswlib: Semantic similarity search
    Combined: Reciprocal Rank Fusion for best results
    """

    EMBEDDING_DIM = 384

    def __init__(self, db_path: str = "/app/data/outdoor/knowledge.db"):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self.index_path = self.db_path.with_suffix(".idx")

        self._conn: sqlite3.Connection | None = None
        self._index = None  # Hnswlib index

        self._init_database()

    def _init_database(self):
        """Initialize SQLite database with FTS5."""
        self._conn = sqlite3.connect(str(self.db_path))
        self._conn.row_factory = sqlite3.Row

        self._conn.executescript("""
            CREATE TABLE IF NOT EXISTS entries (
                id TEXT PRIMARY KEY,
                title TEXT NOT NULL,
                content TEXT NOT NULL,
                category TEXT NOT NULL,
                subcategory TEXT,
                safety_level TEXT DEFAULT 'safe',
                safety_notes TEXT,
                source_file TEXT,
                source_page INTEGER,
                source_url TEXT,
                license TEXT,
                tags TEXT,
                keywords TEXT,
                created_at TEXT,
                updated_at TEXT
            );

            CREATE VIRTUAL TABLE IF NOT EXISTS entries_fts USING fts5(
                id, title, content, tags, keywords,
                content='entries',
                content_rowid='rowid'
            );

            CREATE TRIGGER IF NOT EXISTS entries_ai AFTER INSERT ON entries BEGIN
                INSERT INTO entries_fts(rowid, id, title, content, tags, keywords)
                VALUES (new.rowid, new.id, new.title, new.content, new.tags, new.keywords);
            END;

            CREATE TRIGGER IF NOT EXISTS entries_ad AFTER DELETE ON entries BEGIN
                INSERT INTO entries_fts(entries_fts, rowid, id, title, content, tags, keywords)
                VALUES ('delete', old.rowid, old.id, old.title, old.content, old.tags, old.keywords);
            END;
        """)
        self._conn.commit()

    def _init_vector_index(self, num_elements: int):
        """Initialize Hnswlib index."""
        import hnswlib

        self._index = hnswlib.Index(space='cosine', dim=self.EMBEDDING_DIM)
        self._index.init_index(
            max_elements=num_elements,
            ef_construction=100,
            M=16
        )
        self._index.set_ef(50)

    def add_entries(self, entries: list[KnowledgeEntry]):
        """Add entries to both SQLite and vector index."""
        if not entries:
            return

        # Initialize vector index if needed
        if self._index is None:
            self._init_vector_index(len(entries) + 10000)

        # Insert into SQLite
        for entry in entries:
            self._conn.execute("""
                INSERT OR REPLACE INTO entries
                (id, title, content, category, subcategory, safety_level, safety_notes,
                 source_file, source_page, source_url, license, tags, keywords,
                 created_at, updated_at)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                entry.id, entry.title, entry.content, entry.category.value,
                entry.subcategory, entry.safety_level.value, entry.safety_notes,
                entry.source_file, entry.source_page, entry.source_url, entry.license,
                json.dumps(entry.tags), json.dumps(entry.keywords),
                entry.created_at, entry.updated_at
            ))

        self._conn.commit()

        # Get rowids for vector index mapping
        cursor = self._conn.execute(
            f"SELECT rowid, id FROM entries WHERE id IN ({','.join('?' * len(entries))})",
            [e.id for e in entries]
        )
        id_to_rowid = {row["id"]: row["rowid"] for row in cursor}

        # Add to vector index
        embeddings = [e.embedding for e in entries if e.embedding]
        labels = [id_to_rowid[e.id] for e in entries if e.embedding]

        if embeddings:
            self._index.add_items(embeddings, labels)

    def search(
        self,
        query: str,
        query_embedding: list[float] | None = None,
        category: str | None = None,
        safety_max: str | None = None,
        limit: int = 10,
    ) -> list[dict[str, Any]]:
        """Hybrid search combining FTS and vector similarity."""

        # FTS5 search
        fts_results = self._fts_search(query, category, limit * 2)

        # Vector search
        vector_results = []
        if query_embedding and self._index:
            vector_results = self._vector_search(query_embedding, category, limit * 2)

        # Combine with Reciprocal Rank Fusion
        combined = self._reciprocal_rank_fusion(fts_results, vector_results, limit)

        # Filter by safety level if specified
        if safety_max:
            safety_order = ["safe", "caution", "warning", "danger", "lethal"]
            max_idx = safety_order.index(safety_max)
            combined = [r for r in combined if safety_order.index(r["safety_level"]) <= max_idx]

        return combined[:limit]

    def _fts_search(self, query: str, category: str | None, limit: int) -> list[dict]:
        """Full-text search using FTS5."""
        sql = """
            SELECT e.*, bm25(entries_fts) as score
            FROM entries_fts
            JOIN entries e ON entries_fts.rowid = e.rowid
            WHERE entries_fts MATCH ?
        """
        params = [query]

        if category:
            sql += " AND e.category = ?"
            params.append(category)

        sql += " ORDER BY score LIMIT ?"
        params.append(limit)

        cursor = self._conn.execute(sql, params)
        return [dict(row) for row in cursor.fetchall()]

    def _vector_search(
        self,
        embedding: list[float],
        category: str | None,
        limit: int
    ) -> list[dict]:
        """Semantic search using Hnswlib."""
        labels, distances = self._index.knn_query([embedding], k=limit)

        results = []
        for label, dist in zip(labels[0], distances[0]):
            cursor = self._conn.execute(
                "SELECT * FROM entries WHERE rowid = ?", (int(label),)
            )
            row = cursor.fetchone()
            if row:
                entry = dict(row)
                if category and entry["category"] != category:
                    continue
                entry["vector_score"] = 1 - dist
                results.append(entry)

        return results

    def _reciprocal_rank_fusion(
        self,
        fts_results: list[dict],
        vector_results: list[dict],
        limit: int,
        k: int = 60
    ) -> list[dict]:
        """Combine results using RRF algorithm."""
        scores: dict[str, float] = {}
        entries: dict[str, dict] = {}

        for rank, entry in enumerate(fts_results):
            entry_id = entry["id"]
            scores[entry_id] = scores.get(entry_id, 0) + 1 / (k + rank + 1)
            entries[entry_id] = entry

        for rank, entry in enumerate(vector_results):
            entry_id = entry["id"]
            scores[entry_id] = scores.get(entry_id, 0) + 1 / (k + rank + 1)
            entries[entry_id] = entry

        sorted_ids = sorted(scores.keys(), key=lambda x: scores[x], reverse=True)
        return [entries[eid] for eid in sorted_ids[:limit]]

    def save_index(self):
        """Save vector index to disk."""
        if self._index:
            self._index.save_index(str(self.index_path))

    def load_index(self):
        """Load vector index from disk."""
        import hnswlib

        if self.index_path.exists():
            self._index = hnswlib.Index(space='cosine', dim=self.EMBEDDING_DIM)
            self._index.load_index(str(self.index_path))
            self._index.set_ef(50)

    def get_stats(self) -> dict[str, Any]:
        """Get index statistics."""
        cursor = self._conn.execute("SELECT COUNT(*) FROM entries")
        total = cursor.fetchone()[0]

        cursor = self._conn.execute(
            "SELECT category, COUNT(*) FROM entries GROUP BY category"
        )
        by_category = dict(cursor.fetchall())

        return {
            "total_entries": total,
            "by_category": by_category,
            "db_size_mb": self.db_path.stat().st_size / (1024 * 1024) if self.db_path.exists() else 0,
            "index_size_mb": self.index_path.stat().st_size / (1024 * 1024) if self.index_path.exists() else 0,
        }
```

### Build Pipeline Script

```python
#!/usr/bin/env python3
"""Build the complete RAG knowledge base."""

from __future__ import annotations

import asyncio
import json
from pathlib import Path

from device.services.ai_service.rag.embeddings import EmbeddingGenerator
from device.services.ai_service.rag.models import Category
from device.services.ai_service.rag.parsers.pdf import PDFParser
from device.services.ai_service.rag.parsers.plants import PlantDataParser
from device.services.ai_service.rag.store import HybridKnowledgeStore


async def build_full_index(
    raw_dir: Path = Path("data/raw"),
    output_dir: Path = Path("data/outdoor"),
):
    """Build the complete knowledge base."""

    print("=== Building RAG Knowledge Base ===\n")

    pdf_parser = PDFParser()
    plant_parser = PlantDataParser()
    embedder = EmbeddingGenerator()
    store = HybridKnowledgeStore(str(output_dir / "knowledge.db"))

    all_entries = []

    # 1. Parse survival manuals
    print("📚 Parsing survival manuals...")
    for pdf_file in (raw_dir / "survival").glob("*.pdf"):
        entries = pdf_parser.parse_file(pdf_file, Category.SURVIVAL)
        all_entries.extend(entries)
        print(f"   {pdf_file.name}: {len(entries)} entries")

    # 2. Parse navigation manuals
    print("🧭 Parsing navigation resources...")
    for pdf_file in (raw_dir / "navigation").glob("*.pdf"):
        entries = pdf_parser.parse_file(pdf_file, Category.NAVIGATION)
        all_entries.extend(entries)
        print(f"   {pdf_file.name}: {len(entries)} entries")

    # 3. Parse first aid guides
    print("🏥 Parsing first aid guides...")
    for pdf_file in (raw_dir / "first_aid").glob("*.pdf"):
        entries = pdf_parser.parse_file(pdf_file, Category.FIRST_AID)
        all_entries.extend(entries)
        print(f"   {pdf_file.name}: {len(entries)} entries")

    # 4. Parse plant databases
    print("🌿 Parsing plant databases...")
    for json_file in (raw_dir / "plants").glob("*.json"):
        entries = plant_parser.parse_pfaf_json(json_file)
        all_entries.extend(entries)
        print(f"   {json_file.name}: {len(entries)} entries")

    for csv_file in (raw_dir / "plants").glob("*.csv"):
        entries = plant_parser.parse_usda_csv(csv_file)
        all_entries.extend(entries)
        print(f"   {csv_file.name}: {len(entries)} entries")

    print(f"\n📊 Total entries: {len(all_entries)}")

    # 5. Generate embeddings
    print("\n🔢 Generating embeddings (this may take several minutes)...")
    all_entries = embedder.embed_entries(all_entries)

    # 6. Build hybrid index
    print("\n💾 Building hybrid index...")
    store.add_entries(all_entries)
    store.save_index()

    # 7. Create backup
    backup_path = output_dir / "knowledge_backup.json"
    with open(backup_path, "w") as f:
        json.dump([e.to_dict() for e in all_entries], f)
    print(f"   Backup saved to {backup_path}")

    # 8. Print stats
    stats = store.get_stats()
    print(f"\n✅ Build complete!")
    print(f"   Total entries: {stats['total_entries']}")
    print(f"   DB size: {stats['db_size_mb']:.1f} MB")
    print(f"   Index size: {stats['index_size_mb']:.1f} MB")
    print(f"   Categories: {stats['by_category']}")


if __name__ == "__main__":
    asyncio.run(build_full_index())
```
