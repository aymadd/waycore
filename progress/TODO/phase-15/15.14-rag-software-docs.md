## Description

Implement a Retrieval-Augmented Generation (RAG) system that indexes and
searches the Waycore software documentation, enabling the AI to answer questions
about how to use the device and its features.

## Current State (Problems)

1. **No documentation access**: AI cannot read software manuals
2. **Generic answers**: AI gives general advice, not Waycore-specific
3. **Outdated knowledge**: AI's training data may not include Waycore details
4. **User frustration**: Users ask "How do I..." and get unhelpful responses

## Target State (Goals)

1. **Documentation search**: AI can find relevant docs for any question
2. **Accurate answers**: Responses reference actual Waycore features
3. **Auto-updating**: RAG index updates when docs change
4. **Source citations**: AI can reference specific doc sections

## Implementation

### Document Chunker

```python
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path


@dataclass
class DocChunk:
    """A chunk of documentation."""

    content: str
    source: str  # File path
    section: str  # Section heading
    chunk_id: str


class MarkdownChunker:
    """Chunk markdown documents by sections."""

    def __init__(self, chunk_size: int = 500, overlap: int = 50) -> None:
        self.chunk_size = chunk_size
        self.overlap = overlap

    def chunk_file(self, path: Path) -> list[DocChunk]:
        """Chunk a markdown file."""
        content = path.read_text()
        sections = self._split_by_headers(content)

        chunks = []
        for section_title, section_content in sections:
            section_chunks = self._chunk_text(section_content)
            for i, chunk_text in enumerate(section_chunks):
                chunks.append(DocChunk(
                    content=chunk_text,
                    source=str(path),
                    section=section_title,
                    chunk_id=f"{path.stem}_{section_title}_{i}",
                ))

        return chunks

    def _split_by_headers(self, content: str) -> list[tuple[str, str]]:
        """Split content by markdown headers."""
        pattern = r'^(#{1,3})\s+(.+)$'
        sections = []
        current_title = "Introduction"
        current_content = []

        for line in content.split('\n'):
            match = re.match(pattern, line)
            if match:
                if current_content:
                    sections.append((current_title, '\n'.join(current_content)))
                current_title = match.group(2).strip()
                current_content = []
            else:
                current_content.append(line)

        if current_content:
            sections.append((current_title, '\n'.join(current_content)))

        return sections

    def _chunk_text(self, text: str) -> list[str]:
        """Chunk text into overlapping pieces."""
        words = text.split()
        chunks = []

        for i in range(0, len(words), self.chunk_size - self.overlap):
            chunk = ' '.join(words[i:i + self.chunk_size])
            if chunk.strip():
                chunks.append(chunk)

        return chunks
```

### Embeddings (Local)

```python
from __future__ import annotations

from typing import Any

import numpy as np


class LocalEmbeddings:
    """Generate embeddings using a local model."""

    def __init__(self, model_path: str | None = None) -> None:
        # Use sentence-transformers or similar
        # For Raspberry Pi: use a small model like all-MiniLM-L6-v2
        self._model = None  # Lazy load

    def _load_model(self) -> Any:
        """Load embedding model."""
        if self._model is None:
            from sentence_transformers import SentenceTransformer
            # Small, efficient model suitable for RPi
            self._model = SentenceTransformer('all-MiniLM-L6-v2')
        return self._model

    def embed(self, texts: list[str]) -> list[list[float]]:
        """Generate embeddings for texts."""
        model = self._load_model()
        embeddings = model.encode(texts, convert_to_numpy=True)
        return embeddings.tolist()

    def embed_query(self, query: str) -> list[float]:
        """Generate embedding for a query."""
        return self.embed([query])[0]
```

### Vector Store (ChromaDB)

```python
from __future__ import annotations

from pathlib import Path
from typing import Any

import chromadb
from chromadb.config import Settings

from .chunker import DocChunk
from .embeddings import LocalEmbeddings


class VectorStore:
    """Vector database for document storage and retrieval."""

    def __init__(self, persist_dir: str = "/app/data/rag") -> None:
        self._client = chromadb.Client(Settings(
            persist_directory=persist_dir,
            anonymized_telemetry=False,
        ))
        self._embeddings = LocalEmbeddings()
        self._collection = self._client.get_or_create_collection(
            name="waycore_docs",
            metadata={"hnsw:space": "cosine"},
        )

    def add_chunks(self, chunks: list[DocChunk]) -> None:
        """Add document chunks to the store."""
        if not chunks:
            return

        texts = [c.content for c in chunks]
        embeddings = self._embeddings.embed(texts)

        self._collection.add(
            ids=[c.chunk_id for c in chunks],
            embeddings=embeddings,
            documents=texts,
            metadatas=[{"source": c.source, "section": c.section} for c in chunks],
        )

    def search(self, query: str, n_results: int = 5) -> list[dict[str, Any]]:
        """Search for relevant chunks."""
        query_embedding = self._embeddings.embed_query(query)

        results = self._collection.query(
            query_embeddings=[query_embedding],
            n_results=n_results,
            include=["documents", "metadatas", "distances"],
        )

        return [
            {
                "content": doc,
                "source": meta["source"],
                "section": meta["section"],
                "score": 1 - dist,  # Convert distance to similarity
            }
            for doc, meta, dist in zip(
                results["documents"][0],
                results["metadatas"][0],
                results["distances"][0],
            )
        ]
```

### MCP RAG Server

```python
from __future__ import annotations

from mcp.server import Server
from mcp.types import Tool, TextContent

from device.services.ai_service.rag.retriever import VectorStore

app = Server("waycore-rag-docs")
store = VectorStore()


@app.list_tools()
async def list_tools() -> list[Tool]:
    return [
        Tool(
            name="search_documentation",
            description=(
                "Search Waycore software documentation and user manual. "
                "Use this to answer questions about how to use device features."
            ),
            inputSchema={
                "type": "object",
                "properties": {
                    "query": {
                        "type": "string",
                        "description": "The search query about Waycore features or usage",
                    },
                    "max_results": {
                        "type": "integer",
                        "description": "Maximum number of results (default: 3)",
                        "default": 3,
                    },
                },
                "required": ["query"],
            },
        ),
    ]


@app.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    if name != "search_documentation":
        return [TextContent(type="text", text=f"Unknown tool: {name}")]

    query = arguments.get("query", "")
    max_results = arguments.get("max_results", 3)

    results = store.search(query, n_results=max_results)

    if not results:
        return [TextContent(
            type="text",
            text="No relevant documentation found for this query."
        )]

    formatted = ["Relevant documentation:\n"]
    for i, r in enumerate(results, 1):
        formatted.append(f"\n--- Source: {r['source']} ({r['section']}) ---\n")
        formatted.append(r["content"])
        formatted.append("\n")

    return [TextContent(type="text", text="".join(formatted))]
```

## Index Build Process

```bash
#!/bin/bash
# scripts/build-rag-index.sh

echo "Building RAG index for documentation..."

python -c "
from pathlib import Path
from device.services.ai_service.rag.chunker import MarkdownChunker
from device.services.ai_service.rag.retriever import VectorStore

chunker = MarkdownChunker()
store = VectorStore()

docs_dir = Path('docs')
for md_file in docs_dir.rglob('*.md'):
    print(f'Indexing: {md_file}')
    chunks = chunker.chunk_file(md_file)
    store.add_chunks(chunks)
    print(f'  Added {len(chunks)} chunks')

print('Index build complete!')
"
```
