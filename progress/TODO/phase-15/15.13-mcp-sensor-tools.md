## Description

Create an MCP server that exposes device sensor data as tools for the AI agent. This enables the AI to answer questions like "What's the temperature?" or "Where am I?" with real device data.

## Current State (Problems)

1. **No sensor access**: AI cannot read any sensor data
2. **Generic answers**: AI gives hypothetical answers instead of real data
3. **No context**: AI doesn't know user's location, heading, or environment

## Target State (Goals)

1. **Real-time sensors**: AI can read temperature, GPS, altitude, compass, etc.
2. **Natural queries**: "What's the temperature?" → "Currently 23°C"
3. **Location awareness**: AI knows where the user is
4. **Power status**: AI can report battery level and charging status

## Available Sensors

| Sensor | Tool Name | Description |
|--------|-----------|-------------|
| Temperature | `get_temperature` | Current temp in °C/°F |
| GPS | `get_location` | Lat/long, altitude, accuracy |
| Compass | `get_compass_heading` | Heading in degrees, cardinal direction |
| Altimeter | `get_altitude` | Barometric altitude |
| Battery | `get_battery_status` | Level %, charging state, time remaining |
| Time | `get_device_time` | Current time, date, timezone |

### MCP Sensor Server

```python
from __future__ import annotations

from mcp.server import Server
from mcp.types import Tool, TextContent

app = Server("waycore-sensors")


@app.list_tools()
async def list_tools() -> list[Tool]:
    """List available sensor tools."""
    return [
        Tool(
            name="get_temperature",
            description="Get current temperature reading from device sensor",
            inputSchema={"type": "object", "properties": {}, "required": []},
        ),
        Tool(
            name="get_location",
            description="Get current GPS coordinates (latitude, longitude, altitude)",
            inputSchema={"type": "object", "properties": {}, "required": []},
        ),
        Tool(
            name="get_compass_heading",
            description="Get current compass heading in degrees and cardinal direction",
            inputSchema={"type": "object", "properties": {}, "required": []},
        ),
        Tool(
            name="get_altitude",
            description="Get current altitude from barometric altimeter",
            inputSchema={"type": "object", "properties": {}, "required": []},
        ),
        Tool(
            name="get_battery_status",
            description="Get battery level and charging status",
            inputSchema={"type": "object", "properties": {}, "required": []},
        ),
        Tool(
            name="get_device_time",
            description="Get current device date and time",
            inputSchema={"type": "object", "properties": {}, "required": []},
        ),
    ]


@app.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    """Handle tool calls."""
    handlers = {
        "get_temperature": get_temperature,
        "get_location": get_location,
        "get_compass_heading": get_compass_heading,
        "get_altitude": get_altitude,
        "get_battery_status": get_battery_status,
        "get_device_time": get_device_time,
    }

    handler = handlers.get(name)
    if not handler:
        return [TextContent(type="text", text=f"Unknown tool: {name}")]

    result = await handler(**arguments)
    return [TextContent(type="text", text=result)]
```

### Tool Handlers

```python
from __future__ import annotations

import httpx
from datetime import datetime


CORE_DAEMON_URL = "http://core-daemon:8000"
SENSOR_HUB_URL = "http://sensor-hub:8001"


async def get_temperature() -> str:
    """Get current temperature."""
    async with httpx.AsyncClient() as client:
        try:
            resp = await client.get(f"{SENSOR_HUB_URL}/api/sensors/temperature")
            if resp.status_code == 200:
                data = resp.json()
                temp_c = data["value"]
                temp_f = (temp_c * 9/5) + 32
                return f"Current temperature: {temp_c:.1f}°C ({temp_f:.1f}°F)"
            return "Temperature sensor unavailable"
        except Exception as e:
            return f"Error reading temperature: {e}"


async def get_location() -> str:
    """Get current GPS location."""
    async with httpx.AsyncClient() as client:
        try:
            resp = await client.get(f"{SENSOR_HUB_URL}/api/sensors/gps")
            if resp.status_code == 200:
                data = resp.json()
                lat = data["latitude"]
                lon = data["longitude"]
                alt = data.get("altitude", "unknown")
                acc = data.get("accuracy", "unknown")
                return (
                    f"Current location:\n"
                    f"  Latitude: {lat:.6f}°\n"
                    f"  Longitude: {lon:.6f}°\n"
                    f"  Altitude: {alt}m\n"
                    f"  Accuracy: ±{acc}m"
                )
            return "GPS unavailable"
        except Exception as e:
            return f"Error reading GPS: {e}"


async def get_compass_heading() -> str:
    """Get current compass heading."""
    async with httpx.AsyncClient() as client:
        try:
            resp = await client.get(f"{SENSOR_HUB_URL}/api/sensors/compass")
            if resp.status_code == 200:
                data = resp.json()
                heading = data["heading"]
                cardinal = _heading_to_cardinal(heading)
                return f"Compass heading: {heading:.0f}° ({cardinal})"
            return "Compass unavailable"
        except Exception as e:
            return f"Error reading compass: {e}"


async def get_altitude() -> str:
    """Get current altitude."""
    async with httpx.AsyncClient() as client:
        try:
            resp = await client.get(f"{SENSOR_HUB_URL}/api/sensors/altimeter")
            if resp.status_code == 200:
                data = resp.json()
                alt_m = data["altitude"]
                alt_ft = alt_m * 3.281
                pressure = data.get("pressure", "unknown")
                return (
                    f"Altitude: {alt_m:.0f}m ({alt_ft:.0f}ft)\n"
                    f"Barometric pressure: {pressure} hPa"
                )
            return "Altimeter unavailable"
        except Exception as e:
            return f"Error reading altimeter: {e}"


async def get_battery_status() -> str:
    """Get battery status."""
    async with httpx.AsyncClient() as client:
        try:
            resp = await client.get(f"{CORE_DAEMON_URL}/api/power/status")
            if resp.status_code == 200:
                data = resp.json()
                pct = data["percentage"]
                charging = data.get("is_charging", False)
                remaining = data.get("estimated_remaining_hours")

                status = "charging" if charging else "discharging"
                msg = f"Battery: {pct}% ({status})"
                if remaining:
                    msg += f"\nEstimated remaining: {remaining:.1f} hours"
                return msg
            return "Battery status unavailable"
        except Exception as e:
            return f"Error reading battery: {e}"


async def get_device_time() -> str:
    """Get current device time."""
    now = datetime.now()
    return (
        f"Current time: {now.strftime('%H:%M:%S')}\n"
        f"Date: {now.strftime('%A, %B %d, %Y')}\n"
        f"Timezone: {now.astimezone().tzname()}"
    )


def _heading_to_cardinal(heading: float) -> str:
    """Convert heading to cardinal direction."""
    directions = ["N", "NNE", "NE", "ENE", "E", "ESE", "SE", "SSE",
                  "S", "SSW", "SW", "WSW", "W", "WNW", "NW", "NNW"]
    idx = round(heading / 22.5) % 16
    return directions[idx]
```
