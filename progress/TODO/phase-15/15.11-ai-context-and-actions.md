## Description

Enable the AI assistant to access real-time device data (sensors, databases, applications) to provide contextual answers, and to perform actions on behalf of the user (save notes, etc.). This implements a "tool use" pattern where the AI can invoke functions to gather information or execute tasks.

## Current State (Problems)

1. **No context awareness**: AI cannot access current sensor readings, location, or device state
2. **No historical data**: AI cannot query past sensor readings, notes, or messages
3. **No action capability**: AI cannot perform actions like saving notes or creating mesh messages
4. **Generic responses**: Without device context, AI gives generic answers instead of personalized ones

## Target State (Goals)

1. **Real-time context**: AI can access current sensor data (temperature, GPS, altitude, compass heading)
2. **Historical queries**: AI can search notes, past locations, sensor history
3. **Cross-app awareness**: AI knows what photos exist, mesh nodes, battery status
4. **Action execution**: AI can save notes, potentially send mesh messages (with confirmation)
5. **Natural interaction**: User asks "What's my current altitude?" and AI responds with actual data

## Use Cases

| User Query | AI Action | Response |
|------------|-----------|----------|
| "What's the temperature?" | Read sensor data | "Current temperature is 23°C" |
| "Where am I?" | Read GPS + reverse geocode | "You're at 45.5°N, 122.6°W near Portland" |
| "What's my altitude?" | Read altimeter | "Current altitude is 1,250m" |
| "How's my battery?" | Read power status | "Battery at 67%, about 4 hours remaining" |
| "Find my note about hiking" | Query notes DB | "Found 2 notes mentioning hiking: ..." |
| "Save a note: Pack sunscreen" | Create note | "Note saved: Pack sunscreen" |
| "Who's on the mesh network?" | Query mesh nodes | "3 nodes online: Alpha, Bravo, Charlie" |

## Implementation

### Tool Definition Schema

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Callable


@dataclass
class Tool:
    """Definition of an AI-callable tool."""

    name: str
    description: str
    parameters: dict[str, Any]  # JSON Schema
    handler: Callable[..., Any]
    requires_confirmation: bool = False
    category: str = "query"  # query, action


# Tool Registry
TOOLS: dict[str, Tool] = {}


def register_tool(tool: Tool) -> None:
    """Register a tool for AI use."""
    TOOLS[tool.name] = tool


def get_tool(name: str) -> Tool | None:
    """Get a registered tool by name."""
    return TOOLS.get(name)
```

        if resp.status_code == 200:

async def get_compass_heading() -> dict:
    """Get current compass heading."""
    async with httpx.AsyncClient() as client:
        resp = await client.get("http://core-daemon:8001/api/sensors/compass")
        if resp.status_code == 200:
            data = resp.json()
            heading = data["heading"]
            direction = _heading_to_cardinal(heading)
            return {
                "heading_degrees": heading,
                "cardinal_direction": direction,
                "timestamp": data["timestamp"],
            }
        return {"error": "Compass unavailable"}


def _heading_to_cardinal(heading: float) -> str:
    """Convert heading to cardinal direction."""
    directions = ["N", "NE", "E", "SE", "S", "SW", "W", "NW"]
    idx = round(heading / 45) % 8
    return directions[idx]

### Tool Executor

```python
from __future__ import annotations

import json
import re
from typing import Any

from device.services.ai_service.tools import TOOLS, get_tool


class ToolExecutor:
    """Executes tool calls from AI responses."""

    TOOL_PATTERN = re.compile(
        r"<tool_call>\s*(\w+)\s*(?:\((.*?)\))?\s*</tool_call>",
        re.DOTALL
    )

    async def execute_tools_in_response(self, response: str) -> tuple[str, list[dict]]:
        """
        Parse AI response for tool calls and execute them.

        Returns:
            Tuple of (response with results, list of tool results)
        """
        results = []

        for match in self.TOOL_PATTERN.finditer(response):
            tool_name = match.group(1)
            args_str = match.group(2) or "{}"

            try:
                args = json.loads(args_str) if args_str.strip() else {}
            except json.JSONDecodeError:
                args = {}

            result = await self._execute_tool(tool_name, args)
            results.append({
                "tool": tool_name,
                "args": args,
                "result": result,
            })

        # Replace tool calls with results in response
        final_response = self._inject_results(response, results)

        return final_response, results

    async def _execute_tool(self, name: str, args: dict) -> Any:
        """Execute a single tool."""
        tool = get_tool(name)
        if not tool:
            return {"error": f"Unknown tool: {name}"}

        try:
            return await tool.handler(**args)
        except Exception as e:
            return {"error": str(e)}

    def _inject_results(self, response: str, results: list[dict]) -> str:
        """Replace tool call placeholders with actual results."""
        for result in results:
            pattern = rf"<tool_call>\s*{result['tool']}\s*.*?</tool_call>"
            formatted = self._format_result(result["result"])
            response = re.sub(pattern, formatted, response, count=1, flags=re.DOTALL)

        return response

    def _format_result(self, result: dict) -> str:
        """Format tool result for human-readable output."""
        if "error" in result:
            return f"[Unable to retrieve: {result['error']}]"

        # Format based on content
        if "temperature_celsius" in result:
            return f"{result['temperature_celsius']}°C"
        if "latitude" in result:
            return f"{result['latitude']:.4f}°, {result['longitude']:.4f}°"
        if "altitude_m" in result:
            return f"{result['altitude_m']}m"
        if "percentage" in result:
            return f"{result['percentage']}%"
        if "notes" in result:
            if result["count"] == 0:
                return "No notes found"
            return "\n".join(f"- {n['content']}" for n in result["notes"])

        return json.dumps(result)
```
