## Implementation

### Download Script

```bash
#!/bin/bash
# scripts/download-rag-sources.sh
# Downloads all public domain resources for Waycore RAG

set -e

DATA_DIR="${1:-data/raw}"
mkdir -p "$DATA_DIR"/{survival,navigation,first_aid,plants,weather,knots,comms}

echo "=== Downloading RAG Knowledge Sources ==="
echo "Target: $DATA_DIR"
echo ""

# === SURVIVAL MANUALS ===
echo "📚 Downloading survival manuals..."

wget -q --show-progress -O "$DATA_DIR/survival/FM21-76_Survival.pdf" \
  "https://archive.org/download/Fm21-76SurvivalManual/FM21-76_SurvivalManual.pdf" || \
  echo "  ⚠️ FM21-76 failed - try manual download"

wget -q --show-progress -O "$DATA_DIR/survival/FM21-76-1_Evasion.pdf" \
  "https://irp.fas.org/doddir/army/fm21-76-1.pdf" || \
  echo "  ⚠️ FM21-76-1 failed"

wget -q --show-progress -O "$DATA_DIR/survival/RangerHandbook.pdf" \
  "https://archive.org/download/ranger-handbook-sh-21-76/ranger-handbook-sh-21-76.pdf" || \
  echo "  ⚠️ Ranger Handbook failed"

# === NAVIGATION ===
echo "🧭 Downloading navigation resources..."

wget -q --show-progress -O "$DATA_DIR/navigation/FM3-25.26_MapReading.pdf" \
  "https://irp.fas.org/doddir/army/fm3-25-26.pdf" || \
  echo "  ⚠️ FM3-25.26 failed"

# === FIRST AID ===
echo "🏥 Downloading first aid guides..."

wget -q --show-progress -O "$DATA_DIR/first_aid/FM4-25.11_FirstAid.pdf" \
  "https://irp.fas.org/doddir/army/fm4-25-11.pdf" || \
  echo "  ⚠️ FM4-25.11 failed"

# === KNOTS ===
echo "🪢 Downloading knot resources..."

wget -q --show-progress -O "$DATA_DIR/knots/FM5-125_Rigging.pdf" \
  "https://www.globalsecurity.org/military/library/policy/army/fm/5-125/fm5-125.pdf" || \
  echo "  ⚠️ FM5-125 failed"

# === WEATHER ===
echo "🌦️ Downloading weather resources..."

wget -q --show-progress -O "$DATA_DIR/weather/NOAA_CloudChart.pdf" \
  "https://www.weather.gov/media/owlie/cloud_chart.pdf" || \
  echo "  ⚠️ Cloud chart failed"

# === SUMMARY ===
echo ""
echo "=== Download Summary ==="
find "$DATA_DIR" -type f -name "*.pdf" | wc -l | xargs echo "PDFs downloaded:"
find "$DATA_DIR" -type f -name "*.json" | wc -l | xargs echo "JSON files:"
find "$DATA_DIR" -type f -name "*.csv" | wc -l | xargs echo "CSV files:"
du -sh "$DATA_DIR" | cut -f1 | xargs echo "Total size:"
echo ""
echo "✅ Download complete. Run parsing script next."
```
