# Task: Data Asset Version Management Infrastructure

**Phase**: Phase 15 - AI App **Task ID**: 15.27 **Status**: COMPLETED **Started**:
Not started **Completed**: 2025-12-27 **Priority**: Medium

## Description

Create infrastructure for managing the RAG knowledge base from
`waycore-knowledge` GitHub releases. This provides version checking, automatic
updates, caching, and fallback mechanisms for the knowledge base.

**Note**: AI models use HuggingFace Hub (see task 15.26) which already provides
versioning, caching, and efficient downloads. This task focuses on the knowledge
base which requires a custom solution.

## Motivation

The knowledge base (SQLite + Hnswlib indexes) doesn't fit HuggingFace's model
paradigm, so we need:

1. **Version tracking**: Know which knowledge base version is deployed
2. **Efficient caching**: Avoid re-downloading unchanged data
3. **Graceful degradation**: Handle network failures
4. **Update checking**: Know when new releases are available

### Phase 2: Knowledge Base Manager

Create `device/libs/knowledge/manager.py`:

```python
from __future__ import annotations

import hashlib
import json
import logging
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from typing import Any

import httpx
import yaml

logger = logging.getLogger(__name__)


@dataclass
class KnowledgeVersion:
    """Tracks installed knowledge base version."""
    version: str
    installed_at: str
    entry_count: int
    sha256_db: str | None = None
    sha256_idx: str | None = None


class KnowledgeBaseManager:
    """Manages the RAG knowledge base from waycore-knowledge releases."""

    REPO = "dmitry-grechko/waycore-knowledge"
    API_URL = f"https://api.github.com/repos/{REPO}/releases"

    def __init__(
        self,
        data_dir: Path | None = None,
        config_path: Path | None = None,
    ):
        self.data_dir = data_dir or Path("data/outdoor")
        self.config_path = config_path or Path("config/knowledge.yaml")
        self.lock_file = self.data_dir.parent / "knowledge.lock"

    def check_for_updates(self) -> str | None:
        """Check if a newer version is available.

        Returns:
            Latest version tag if update available, None if up-to-date
        """
        installed = self.get_installed_version()
        latest = self._fetch_latest_version()

        if installed != latest:
            return latest
        return None

    def download(
        self,
        version: str = "latest",
        force: bool = False,
    ) -> bool:
        """Download knowledge base from GitHub releases.

        Args:
            version: Version tag or "latest"
            force: Re-download even if already installed

        Returns:
            True if download succeeded
        """
        if not force and self._is_current(version):
            logger.info(f"Knowledge base v{version} already installed")
            return True

        try:
            release = self._get_release(version)
            self._download_assets(release)
            self._verify_checksums(release)
            self._update_lock(release)
            return True
        except Exception as e:
            logger.error(f"Failed to download knowledge base: {e}")
            return False

    def get_installed_version(self) -> str | None:
        """Get currently installed version."""
        if not self.lock_file.exists():
            return None
        lock = json.loads(self.lock_file.read_text())
        return lock.get("version")

    def verify_integrity(self) -> bool:
        """Verify integrity of installed knowledge base."""
        db_path = self.data_dir / "knowledge.db"
        idx_path = self.data_dir / "knowledge.idx"

        if not db_path.exists() or not idx_path.exists():
            return False

        # Verify checksums if available
        if self.lock_file.exists():
            lock = json.loads(self.lock_file.read_text())
            if "sha256_db" in lock:
                actual = self._compute_sha256(db_path)
                if actual != lock["sha256_db"]:
                    return False

        return True

    def get_stats(self) -> dict[str, Any]:
        """Get knowledge base statistics."""
        return {
            "version": self.get_installed_version(),
            "db_exists": (self.data_dir / "knowledge.db").exists(),
            "idx_exists": (self.data_dir / "knowledge.idx").exists(),
            "integrity": self.verify_integrity(),
        }

    def _fetch_latest_version(self) -> str:
        """Fetch latest release version from GitHub."""
        response = httpx.get(f"{self.API_URL}/latest")
        response.raise_for_status()
        return response.json()["tag_name"]

    def _compute_sha256(self, path: Path) -> str:
        """Compute SHA256 hash of a file."""
        sha256 = hashlib.sha256()
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(8192), b""):
                sha256.update(chunk)
        return sha256.hexdigest()
```

### Phase 3: CLI Integration

Integrate with existing `scripts/download-knowledge.sh` (from task 15.25):

```bash
#!/bin/bash
# Download knowledge base from waycore-knowledge releases
# Usage: ./scripts/download-knowledge.sh [version]

VERSION="${1:-latest}"

poetry run python -c "
from device.libs.knowledge.manager import KnowledgeBaseManager

manager = KnowledgeBaseManager()

# Check current status
current = manager.get_installed_version()
if current:
    print(f'Current version: {current}')

# Check for updates
if '$VERSION' == 'latest':
    update = manager.check_for_updates()
    if update:
        print(f'Update available: {update}')
    else:
        print('Already up to date')
        exit(0)

# Download
success = manager.download('$VERSION')
if success:
    stats = manager.get_stats()
    print(f'✓ Knowledge base v{stats[\"version\"]} installed')
else:
    print('✗ Download failed')
    exit(1)
"
```
