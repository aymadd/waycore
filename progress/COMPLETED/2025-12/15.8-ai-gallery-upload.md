@Slot(str, result="QVariant")
def classifyImageFromPath(self, file_path: str) -> dict:
    """Load image from file and classify."""
    with open(file_path, "rb") as f:
        image_data = f.read()
    image_b64 = base64.b64encode(image_data).decode()
    return self.classifyImageAndChat(image_b64)
